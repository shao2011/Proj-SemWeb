# query 1: sách và tác giả

PREFIX ex:   <http://example.org/books#>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?book ?title ?author ?authorName
WHERE {
    ?book rdf:type ex:Book ;
          rdfs:label ?title ;
          ex:isWrittenBy ?author .

    ?author rdfs:label ?authorName .
}
LIMIT 20



# query 2: Các sách thuộc 1 series

PREFIX ex:   <http://example.org/books#>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

    ?book rdf:type ex:Book ;
          rdfs:label ?title ;
          ex:isPartOfSeries ?series .

    ?series rdfs:label ?seriesName .
}
LIMIT 20



# query 3: Các sách đã thắng award

PREFIX ex:   <http://example.org/books#>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?book ?title ?award ?awardName ?year
WHERE {
    ?book rdf:type ex:Book ;
          rdfs:label ?title ;
          ex:hasAwardRecognition ?recognition .

    ?recognition rdf:type ex:AwardWin ;
                 ex:forAward ?award .

    OPTIONAL {
        ?recognition ex:awardYear ?year .
    }

    ?award rdfs:label ?awardName .
}
LIMIT 20



# query 4: các edition đã được translate và translator tương ứng
PREFIX ex:   <http://example.org/books#>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?book ?title ?edition ?translator ?translatorName
WHERE {
    ?book rdf:type ex:Book ;
          ex:hasEdition ?edition ;
          rdfs:label ?title .

    ?edition ex:isTranslatedBy ?translator .

    ?translator rdfs:label ?translatorName .
}
LIMIT 20



# query 5: top sách rating cao nhưng chỉ xét có nhiều lượt đánh giá

PREFIX ex:   <http://example.org/books#>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?book ?title ?rating ?numRatings
WHERE {
    ?book rdf:type ex:Book ;
          rdfs:label ?title ;
          ex:rating ?rating ;
          ex:numRatings ?numRatings .

    FILTER(?numRatings >= 10000)
}
ORDER BY DESC(?rating)
LIMIT 10



# query 6: linking

PREFIX ex:   <http://example.org/books#>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX owl:  <http://www.w3.org/2002/07/owl#>

SELECT ?book ?title ?externalResource
WHERE {
    ?book rdf:type ex:Book ;
          rdfs:label ?title ;
          owl:sameAs ?externalResource .
}
LIMIT 20
